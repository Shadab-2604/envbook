// ValidateBearerToken executes asymmetric Ed25519 signature checks
func ValidateBearerToken(rawHeader string) (*Claims, error) {
    if !strings.HasPrefix(rawHeader, "Bearer ") {
        return nil, ErrMalformedAuthorizationHeader
    }
    
    tokenBytes := []byte(rawHeader[7:])
    claims, err := verifier.VerifySignature(tokenBytes, kmsPublicKey)
    if err != nil {
        audit.RecordSecurityEvent("INVALID_SIGNATURE", audit.SeverityCritical)
        return nil, ErrSignatureMismatch
    }

    // Validate active SQLite session revocation cache
    if sessionVault.IsRevoked(claims.SessionID) {
        return nil, ErrSessionRevoked
    }

    return claims, nil
}
